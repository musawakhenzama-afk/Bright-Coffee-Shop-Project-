Task 3: Data Analysis in Excel
---Total Revenue per Product Type
SELECT
    PRODUCT_TYPE,
    SUM(TRANSACTION_QTY * UNIT_PRICE) AS TOTAL_REVENUE
FROM "BRIGHTLEARNCOFFEESHOPGROUP"."PUBLIC"."BRIGHTLEARNSALESDATA"
GROUP BY PRODUCT_TYPE
ORDER BY TOTAL_REVENUE DESC;
  
---Quantity Sold by Product Category

  SELECT
    PRODUCT_CATEGORY,
    SUM(TRANSACTION_QTY) AS TOTAL_QUANTITY,
    SUM(TRANSACTION_QTY * UNIT_PRICE) AS TOTAL_REVENUE
FROM "BRIGHTLEARNCOFFEESHOPGROUP"."PUBLIC"."BRIGHTLEARNSALESDATA"
GROUP BY PRODUCT_CATEGORY
ORDER BY TOTAL_QUANTITY DESC;

---Best-Selling Product Details

SELECT
    PRODUCT_DETAIL,
    SUM(TRANSACTION_QTY) AS TOTAL_QUANTITY,
    SUM(TRANSACTION_QTY * UNIT_PRICE) AS TOTAL_REVENUE
FROM "BRIGHTLEARNCOFFEESHOPGROUP"."PUBLIC"."BRIGHTLEARNSALESDATA"
GROUP BY PRODUCT_DETAIL
ORDER BY TOTAL_QUANTITY DESC
LIMIT 10;

---Revenue per Store Location

SELECT 
    STORE_LOCATION,
    SUM(TRANSACTION_QTY * UNIT_PRICE) AS TOTAL_REVENUE,
    SUM(TRANSACTION_QTY) AS TOTAL_QUANTITY
FROM "BRIGHTLEARNCOFFEESHOPGROUP"."PUBLIC"."BRIGHTLEARNSALESDATA"
GROUP BY STORE_LOCATION
ORDER BY TOTAL_REVENUE DESC;

---Worst-Performing Products by Category

SELECT
    PRODUCT_CATEGORY,
    PRODUCT_DETAIL,
    SUM(TRANSACTION_QTY * UNIT_PRICE) AS TOTAL_REVENUE,
    SUM(TRANSACTION_QTY) AS TOTAL_QUANTITY
FROM "BRIGHTLEARNCOFFEESHOPGROUP"."PUBLIC"."BRIGHTLEARNSALESDATA"
GROUP BY PRODUCT_CATEGORY, PRODUCT_DETAIL
ORDER BY TOTAL_REVENUE ASC
LIMIT 10;

---Time of Day Buckets

SELECT
    TO_CHAR(TRANSACTION_DATE, 'Day') AS DAY_OF_WEEK,
    CASE
        WHEN EXTRACT(HOUR FROM TRANSACTION_TIME::TIME) BETWEEN 5 AND 10 THEN 'Breakfast'
        WHEN EXTRACT(HOUR FROM TRANSACTION_TIME::TIME) BETWEEN 11 AND 12 THEN 'Midday'
        WHEN EXTRACT(HOUR FROM TRANSACTION_TIME::TIME) BETWEEN 13 AND 17 THEN 'Afternoon'
        WHEN EXTRACT(HOUR FROM TRANSACTION_TIME::TIME) BETWEEN 18 AND 20 THEN 'Evening'
        ELSE 'Other'
    END AS TIME_SLOT,
    PRODUCT_CATEGORY,
    PRODUCT_DETAIL,
    SUM(TRANSACTION_QTY * UNIT_PRICE) AS TOTAL_REVENUE,
    SUM(TRANSACTION_QTY) AS TOTAL_QUANTITY
FROM "BRIGHTLEARNCOFFEESHOPGROUP"."PUBLIC"."BRIGHTLEARNSALESDATA"
GROUP BY DAY_OF_WEEK, TIME_SLOT, PRODUCT_CATEGORY, PRODUCT_DETAIL
ORDER BY TOTAL_REVENUE ASC
LIMIT 10; 

---Store Performance by Month
SELECT
    STORE_ID,
    STORE_LOCATION,
    TO_CHAR(TRANSACTION_DATE, 'Month') AS MONTH_NAME,
    SUM(TRANSACTION_QTY * UNIT_PRICE) AS TOTAL_REVENUE,
    SUM(TRANSACTION_QTY) AS TOTAL_QUANTITY
FROM "BRIGHTLEARNCOFFEESHOPGROUP"."PUBLIC"."BRIGHTLEARNSALESDATA"
GROUP BY STORE_ID, STORE_LOCATION, MONTH_NAME
ORDER BY STORE_LOCATION, MONTH_NAME;

---Store Performance by Week

SELECT
    STORE_ID,
    STORE_LOCATION,
    TO_CHAR(TRANSACTION_DATE, 'IW') AS WEEK_NUMBER,
    SUM(TRANSACTION_QTY * UNIT_PRICE) AS TOTAL_REVENUE,
    SUM(TRANSACTION_QTY) AS TOTAL_QUANTITY
FROM "BRIGHTLEARNCOFFEESHOPGROUP"."PUBLIC"."BRIGHTLEARNSALESDATA"
GROUP BY STORE_ID, STORE_LOCATION, WEEK_NUMBER
ORDER BY STORE_LOCATION, WEEK_NUMBER;

--- Day of the Week and Time Slot

SELECT
    STORE_ID,
    STORE_LOCATION,
    TO_CHAR(TRANSACTION_DATE, 'Month') AS MONTH_NAME,
    TO_CHAR(TRANSACTION_DATE, 'IW') AS WEEK_NUMBER,
    TO_CHAR(TRANSACTION_DATE, 'Day') AS DAY_OF_WEEK,
    CASE
        WHEN EXTRACT(HOUR FROM TRANSACTION_TIME::TIME) BETWEEN 5 AND 10 THEN 'Breakfast'
        WHEN EXTRACT(HOUR FROM TRANSACTION_TIME::TIME) BETWEEN 11 AND 12 THEN 'Midday'
        WHEN EXTRACT(HOUR FROM TRANSACTION_TIME::TIME) BETWEEN 13 AND 17 THEN 'Afternoon'
        WHEN EXTRACT(HOUR FROM TRANSACTION_TIME::TIME) BETWEEN 18 AND 20 THEN 'Evening'
        ELSE 'Other'
    END AS TIME_SLOT,
    SUM(TRANSACTION_QTY * UNIT_PRICE) AS TOTAL_REVENUE,
    SUM(TRANSACTION_QTY) AS TOTAL_QUANTITY
FROM "BRIGHTLEARNCOFFEESHOPGROUP"."PUBLIC"."BRIGHTLEARNSALESDATA"
GROUP BY STORE_ID, STORE_LOCATION, MONTH_NAME, WEEK_NUMBER, DAY_OF_WEEK, TIME_SLOT
ORDER BY STORE_LOCATION, MONTH_NAME, WEEK_NUMBER;
